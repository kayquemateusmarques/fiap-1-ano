// Calcule o valor da gorjeta do garçom e o valor da conta

const jantar = parseFloat(prompt('Digite o valor do seu jantar'))
const taxa = jantar * 0.1
const totaldacompra = parseFloat(jantar + taxa)
alert(`Valor inicial da compra: R$ ${jantar.toFixed(2)}\nGorjeta do garçom: R$ ${taxa.toFixed(2)}\nValor total da compra: R$ ${totaldacompra.toFixed(2)}`)
